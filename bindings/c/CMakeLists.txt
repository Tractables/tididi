cmake_minimum_required(VERSION 3.16)
project(tididi_c VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
find_program(CARGO cargo REQUIRED)
set(RUST_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/cargo/release")
if(WIN32)
  set(RUST_LIBRARY "${RUST_OUTPUT}/tididi_c.dll")
else()
  set(RUST_LIBRARY "${RUST_OUTPUT}/${CMAKE_SHARED_LIBRARY_PREFIX}tididi_c${CMAKE_SHARED_LIBRARY_SUFFIX}")
endif()
add_custom_target(tididi_native ALL
  COMMAND "${CMAKE_COMMAND}" -E env CARGO_BUILD_JOBS=8 "${CARGO}" build --release --locked
          --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml"
          --target-dir "${CMAKE_CURRENT_BINARY_DIR}/cargo"
  BYPRODUCTS "${RUST_LIBRARY}" VERBATIM)
add_library(tididi::tididi SHARED IMPORTED GLOBAL)
set_target_properties(tididi::tididi PROPERTIES IMPORTED_LOCATION "${RUST_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(WIN32)
  set_target_properties(tididi::tididi PROPERTIES IMPORTED_IMPLIB "${RUST_OUTPUT}/tididi_c.dll.lib")
endif()
if(APPLE)
  set_target_properties(tididi::tididi PROPERTIES IMPORTED_SONAME "@rpath/libtididi_c.dylib")
elseif(UNIX)
  set_target_properties(tididi::tididi PROPERTIES IMPORTED_NO_SONAME TRUE)
endif()
add_dependencies(tididi::tididi tididi_native)

option(TIDIDI_BUILD_TESTS "Build C and C++ consumer tests" ON)
option(TIDIDI_BUILD_EXAMPLES "Build executable C tutorials" ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_BINARY_DIR}/bin")
endforeach()
function(tididi_consumer name source)
  add_executable(${name} "${source}")
  target_link_libraries(${name} PRIVATE tididi::tididi)
  set_target_properties(${name} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /WX)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(${name} PRIVATE m)
  endif()
  if(WIN32)
    add_custom_command(TARGET ${name} POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${RUST_LIBRARY}" "$<TARGET_FILE_DIR:${name}>")
  endif()
endfunction()
if(TIDIDI_BUILD_TESTS)
  enable_testing()
  tididi_consumer(consumer tests/consumer.c)
  tididi_consumer(cpp_consumer tests/cpp.cpp)
  tididi_consumer(conformance tests/conformance.c)
  add_test(NAME shared_conformance COMMAND conformance "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/conformance.txt")
  tididi_consumer(conformance_sessions tests/conformance_sessions.c)
  add_test(NAME shared_sessions COMMAND conformance_sessions "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/conformance_sessions.txt")
  add_test(NAME c_consumer COMMAND consumer)
  add_test(NAME cpp_consumer COMMAND cpp_consumer)
endif()
if(TIDIDI_BUILD_EXAMPLES)
  file(GLOB examples CONFIGURE_DEPENDS "examples/*.c")
  foreach(example IN LISTS examples)
    get_filename_component(name "${example}" NAME_WE)
    tididi_consumer(${name} "${example}")
    if(TIDIDI_BUILD_TESTS)
      add_test(NAME "example_${name}" COMMAND "${name}")
    endif()
  endforeach()
endif()

install(FILES include/tididi.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
if(WIN32)
  install(FILES "${RUST_LIBRARY}" DESTINATION "${CMAKE_INSTALL_BINDIR}")
  install(FILES "${RUST_OUTPUT}/tididi_c.dll.lib" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
else()
  install(FILES "${RUST_LIBRARY}" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
endif()
install(FILES ../../LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/tididi")
include(CMakePackageConfigHelpers)
configure_package_config_file(tididiConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/tididiConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/tididi"
  PATH_VARS CMAKE_INSTALL_INCLUDEDIR CMAKE_INSTALL_LIBDIR CMAKE_INSTALL_BINDIR)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/tididiConfigVersion.cmake"
  VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/tididiConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/tididiConfigVersion.cmake"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/tididi")
